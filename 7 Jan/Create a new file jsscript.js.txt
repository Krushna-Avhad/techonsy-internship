// ================= MOBILE MENU =================
function initializeMobileMenu() {
    const navbar = document.querySelector('.navbar');
    const navLinks = document.querySelector('.nav-links');
    
    if (!navbar) return;
    
    // Create menu toggle button
    const menuToggle = document.createElement('button');
    menuToggle.className = 'menu-toggle';
    menuToggle.innerHTML = '☰';
    menuToggle.setAttribute('aria-label', 'Toggle menu');
    
    // Insert toggle button
    navbar.insertBefore(menuToggle, navLinks);
    
    // Toggle menu on click
    menuToggle.addEventListener('click', function() {
        navLinks.classList.toggle('active');
        menuToggle.innerHTML = navLinks.classList.contains('active') ? '✕' : '☰';
    });
    
    // Close menu when clicking outside
    document.addEventListener('click', function(event) {
        if (!navbar.contains(event.target) && navLinks.classList.contains('active')) {
            navLinks.classList.remove('active');
            menuToggle.innerHTML = '☰';
        }
    });
    
    // Close menu on window resize
    window.addEventListener('resize', function() {
        if (window.innerWidth > 768 && navLinks.classList.contains('active')) {
            navLinks.classList.remove('active');
            menuToggle.innerHTML = '☰';
        }
    });
}

// ================= BACK TO TOP BUTTON =================
function createBackToTopButton() {
    const backToTopBtn = document.createElement('button');
    backToTopBtn.className = 'back-to-top';
    backToTopBtn.innerHTML = '↑';
    backToTopBtn.setAttribute('aria-label', 'Back to top');
    
    // Add styles
    Object.assign(backToTopBtn.style, {
        position: 'fixed',
        bottom: '30px',
        right: '30px',
        background: 'var(--brand-primary)',
        color: 'var(--brand-gold)',
        border: 'none',
        borderRadius: '50%',
        width: '50px',
        height: '50px',
        fontSize: '24px',
        cursor: 'pointer',
        display: 'none',
        zIndex: '1000',
        boxShadow: '0 4px 15px rgba(0,0,0,0.3)',
        transition: 'all 0.3s ease'
    });
    
    document.body.appendChild(backToTopBtn);
    
    // Show/hide button based on scroll
    window.addEventListener('scroll', function() {
        if (window.pageYOffset > 500) {
            backToTopBtn.style.display = 'block';
            setTimeout(() => {
                backToTopBtn.style.opacity = '1';
                backToTopBtn.style.transform = 'scale(1)';
            }, 10);
        } else {
            backToTopBtn.style.opacity = '0';
            backToTopBtn.style.transform = 'scale(0.8)';
            setTimeout(() => {
                if (window.pageYOffset <= 500) {
                    backToTopBtn.style.display = 'none';
                }
            }, 300);
        }
    });
    
    // Smooth scroll to top
    backToTopBtn.addEventListener('click', function() {
        window.scrollTo({
            top: 0,
            behavior: 'smooth'
        });
    });
    
    // Hover effect
    backToTopBtn.addEventListener('mouseenter', function() {
        this.style.background = 'var(--brand-gold)';
        this.style.color = 'var(--brand-primary)';
        this.style.transform = 'scale(1.1)';
    });
    
    backToTopBtn.addEventListener('mouseleave', function() {
        this.style.background = 'var(--brand-primary)';
        this.style.color = 'var(--brand-gold)';
        this.style.transform = 'scale(1)';
    });
}

// ================= CITY CARDS ENHANCEMENT =================
function enhanceCityCards() {
    const cityCards = document.querySelectorAll('.city-card');
    
    cityCards.forEach(card => {
        // Add click effect
        card.addEventListener('click', function(e) {
            // Prevent if clicking on button or link inside
            if (e.target.tagName === 'BUTTON' || e.target.tagName === 'A') return;
            
            // Find the parent link
            const link = this.closest('.city-link') || this.querySelector('a');
            if (link) {
                window.location.href = link.href;
            }
        });
        
        // Enhanced hover effects
        card.addEventListener('mouseenter', function() {
            this.style.transform = 'translateY(-12px) scale(1.02)';
            this.style.boxShadow = '0 25px 50px rgba(0,0,0,0.2)';
            
            // Add shimmer effect
            const img = this.querySelector('img');
            if (img) {
                img.style.transform = 'scale(1.1)';
            }
        });
        
        card.addEventListener('mouseleave', function() {
            this.style.transform = 'translateY(0) scale(1)';
            this.style.boxShadow = '';
            
            const img = this.querySelector('img');
            if (img) {
                img.style.transform = 'scale(1)';
            }
        });
        
        // Button hover effect
        const button = card.querySelector('button');
        if (button) {
            button.addEventListener('mouseenter', function(e) {
                e.stopPropagation();
                this.style.transform = 'scale(1.1)';
            });
            
            button.addEventListener('mouseleave', function(e) {
                e.stopPropagation();
                this.style.transform = 'scale(1)';
            });
        }
    });
}

// ================= FORM VALIDATIONS =================
function initializeFormValidations() {
    // Newsletter form
    const newsletterForm = document.querySelector('.footer-section form');
    if (newsletterForm) {
        newsletterForm.addEventListener('submit', function(e) {
            e.preventDefault();
            const emailInput = this.querySelector('input[type="email"]');
            const email = emailInput.value.trim();
            
            if (validateEmail(email)) {
                // Show success
                showFormMessage(this, 'Thank you for subscribing!', 'success');
                emailInput.value = '';
                
                // Save to localStorage
                saveSubscriber(email);
            } else {
                showFormMessage(this, 'Please enter a valid email address.', 'error');
                emailInput.focus();
            }
        });
    }
    
    // Contact form (if exists)
    const contactForm = document.querySelector('.contact-form');
    if (contactForm) {
        contactForm.addEventListener('submit', function(e) {
            e.preventDefault();
            
            const name = this.querySelector('input[name="name"]')?.value.trim();
            const email = this.querySelector('input[type="email"]')?.value.trim();
            const message = this.querySelector('textarea')?.value.trim();
            
            let isValid = true;
            
            // Validate name
            if (!name || name.length < 2) {
                showFormMessage(this, 'Please enter a valid name (min 2 characters).', 'error');
                isValid = false;
            }
            
            // Validate email
            if (!validateEmail(email)) {
                showFormMessage(this, 'Please enter a valid email address.', 'error');
                isValid = false;
            }
            
            // Validate message
            if (!message || message.length < 10) {
                showFormMessage(this, 'Please enter a message (min 10 characters).', 'error');
                isValid = false;
            }
            
            if (isValid) {
                showFormMessage(this, 'Thank you! Your message has been sent.', 'success');
                
                // Simulate form submission
                const formData = {
                    name,
                    email,
                    message,
                    timestamp: new Date().toISOString()
                };
                
                // Save to localStorage
                saveContactMessage(formData);
                
                // Reset form
                this.reset();
            }
        });
    }
}

function validateEmail(email) {
    const re = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return re.test(email);
}

function showFormMessage(form, message, type) {
    // Remove existing messages
    const existingMsg = form.querySelector('.form-message');
    if (existingMsg) existingMsg.remove();
    
    // Create new message
    const msgDiv = document.createElement('div');
    msgDiv.className = `form-message ${type}-message`;
    msgDiv.textContent = message;
    msgDiv.style.cssText = `
        margin-top: 15px;
        padding: 12px;
        border-radius: 8px;
        font-size: 14px;
        text-align: center;
        animation: fadeIn 0.3s ease;
    `;
    
    if (type === 'success') {
        msgDiv.style.backgroundColor = 'rgba(35, 209, 96, 0.1)';
        msgDiv.style.color = '#23d160';
        msgDiv.style.border = '1px solid #23d160';
    } else {
        msgDiv.style.backgroundColor = 'rgba(255, 56, 96, 0.1)';
        msgDiv.style.color = '#ff3860';
        msgDiv.style.border = '1px solid #ff3860';
    }
    
    form.appendChild(msgDiv);
    
    // Remove after 5 seconds
    setTimeout(() => {
        if (msgDiv.parentNode === form) {
            msgDiv.style.opacity = '0';
            msgDiv.style.transition = 'opacity 0.3s ease';
            setTimeout(() => msgDiv.remove(), 300);
        }
    }, 5000);
}

async function initializeWeatherWidget() {
    const cityName = document.querySelector('.city-hero h1')?.textContent;
    if (!cityName) return;
    
    // Map city names to OpenWeatherMap city IDs
    const cityMap = {
        'Mumbai': '1275339',
        'Pune': '1259229',
        'Mysore': '1262321',
        'Jaipur': '1269515',
        'Varanasi': '1253405'
    };
    
    const cityKey = Object.keys(cityMap).find(key => cityName.includes(key));
    if (!cityKey) return;
    
    const weatherDiv = document.createElement('div');
    weatherDiv.className = 'weather-widget';
    weatherDiv.innerHTML = `
        <div class="weather-loading">Loading weather...</div>
        <div class="weather-content" style="display: none;">
            <h4>Current Weather</h4>
            <div class="weather-info">
                <span class="weather-temp">--°C</span>
                <span class="weather-desc">--</span>
                <span class="weather-humidity">Humidity: --%</span>
            </div>
        </div>
    `;
    
    // Add styles
    Object.assign(weatherDiv.style, {
        position: 'absolute',
        top: '20px',
        right: '20px',
        background: 'rgba(255, 255, 255, 0.9)',
        padding: '15px',
        borderRadius: '10px',
        boxShadow: '0 4px 15px rgba(0,0,0,0.2)',
        zIndex: '100',
        maxWidth: '200px',
        fontSize: '14px'
    });
    
    const hero = document.querySelector('.city-hero');
    if (hero) {
        hero.style.position = 'relative';
        hero.appendChild(weatherDiv);
        
        try {
            // Using a free weather API (OpenWeatherMap)
            // Replace with your own API key or use a free alternative
            const apiKey = 'YOUR_API_KEY_HERE'; // You need to sign up for a free key
            const cityId = cityMap[cityKey];
            
            if (apiKey !== 'YOUR_API_KEY_HERE') {
                const response = await fetch(
                    `https://api.openweathermap.org/data/2.5/weather?id=${cityId}&units=metric&appid=${apiKey}`
                );
                
                if (response.ok) {
                    const data = await response.json();
                    
                    const temp = Math.round(data.main.temp);
                    const desc = data.weather[0].description;
                    const humidity = data.main.humidity;
                    const icon = data.weather[0].icon;
                    
                    weatherDiv.querySelector('.weather-loading').style.display = 'none';
                    const content = weatherDiv.querySelector('.weather-content');
                    content.style.display = 'block';
                    
                    content.innerHTML = `
                        <h4 style="margin: 0 0 10px 0; color: var(--brand-primary);">${cityKey} Weather</h4>
                        <div style="display: flex; align-items: center; gap: 10px;">
                            <img src="https://openweathermap.org/img/wn/${icon}.png" alt="${desc}" style="width: 40px; height: 40px;">
                            <div>
                                <div style="font-size: 24px; font-weight: bold;">${temp}°C</div>
                                <div style="text-transform: capitalize;">${desc}</div>
                                <div>Humidity: ${humidity}%</div>
                            </div>
                        </div>
                    `;
                }
            }
        } catch (error) {
            console.log('Weather API error:', error);
            weatherDiv.querySelector('.weather-loading').textContent = 'Weather unavailable';
        }
    }
}

            const budget = parseInt(document.getElementById("budget")?.value);
            const type = document.getElementById("travelType")?.value;
            
            if (!days || !budget || days < 1 || budget < 1000) {
                alert("Please enter valid days (≥1) and budget (≥1000)");
                return;
            }
            
            const suitableCities = cityData.filter(city =>
                city.type === type &&
                days >= city.minDays &&
                budget >= city.minBudget
            );
            
            // Sort by best match (closest to budget)
            suitableCities.sort((a, b) => {
                const aDiff = budget - a.minBudget;
                const bDiff = budget - b.minBudget;
                return Math.abs(aDiff) - Math.abs(bDiff);
            });
            
            displayTripResults(suitableCities, days, budget, type);
            
            // Save to localStorage
            saveTripPlan({ days, budget, type, cities: suitableCities });
        });
    }

function initializeImageGallery() {

    galleryItems.forEach(item => {
        item.addEventListener('click', function(e) {
            e.preventDefault();
            const imgSrc = this.querySelector('img')?.src;

function openLightbox(imgSrc, title) {
    // Create lightbox
    const lightbox = document.createElement('div');
    lightbox.className = 'lightbox';
    lightbox.innerHTML = `
        <div class="lightbox-content">
            <span class="close-lightbox">&times;</span>
            <img src="${imgSrc}" alt="${title || ''}">
            ${title ? `<p>${title}</p>` : ''}
            <div class="lightbox-nav">
                <button class="prev-btn">← Previous</button>
                <button class="next-btn">Next →</button>
            </div>
        </div>
    `;
    
    // Add styles
    Object.assign(lightbox.style, {
        position: 'fixed',
        top: '0',
        left: '0',
        width: '100%',
        height: '100%',
        background: 'rgba(0,0,0,0.9)',
        display: 'flex',
        justifyContent: 'center',
        alignItems: 'center',
        zIndex: '2000',
        animation: 'fadeIn 0.3s ease'
    });
    
    const content = lightbox.querySelector('.lightbox-content');
    Object.assign(content.style, {
        maxWidth: '90%',
        maxHeight: '90%',
        position: 'relative'
    });
    
    lightbox.querySelector('img').style.cssText = `
        max-width: 100%;
        max-height: 70vh;
        border-radius: 10px;
    `;
    
    document.body.appendChild(lightbox);
    document.body.style.overflow = 'hidden';
    
    // Close lightbox
    lightbox.querySelector('.close-lightbox').addEventListener('click', () => {
        document.body.removeChild(lightbox);
        document.body.style.overflow = 'auto';
    });
    
    lightbox.addEventListener('click', (e) => {
        if (e.target === lightbox) {
            document.body.removeChild(lightbox);
            document.body.style.overflow = 'auto';
        }
    });
    
    // Keyboard navigation
    document.addEventListener('keydown', function handleKey(e) {
        if (!document.body.contains(lightbox)) {
            document.removeEventListener('keydown', handleKey);
            return;
        }
        
        if (e.key === 'Escape') {
            document.body.removeChild(lightbox);
            document.body.style.overflow = 'auto';
        }
    });
}

// ================= THEME TOGGLE =================
function initializeThemeToggle() {
    // Check for saved theme
    const savedTheme = localStorage.getItem('theme') || 'light';
    if (savedTheme === 'dark') {
        document.body.classList.add('dark-mode');
    }
    
    // Create toggle button
    const themeToggle = document.createElement('button');
    themeToggle.className = 'theme-toggle';
    themeToggle.innerHTML = savedTheme === 'dark' ? '☀️' : '🌙';
    themeToggle.setAttribute('aria-label', 'Toggle dark mode');
    themeToggle.title = 'Toggle dark/light mode';
    
    Object.assign(themeToggle.style, {
        position: 'fixed',
        bottom: '90px',
        right: '30px',
        background: 'var(--brand-primary)',
        color: 'var(--brand-gold)',
        border: 'none',
        borderRadius: '50%',
        width: '50px',
        height: '50px',
        fontSize: '20px',
        cursor: 'pointer',
        zIndex: '1000',
        boxShadow: '0 4px 15px rgba(0,0,0,0.3)',
        transition: 'all 0.3s ease'
    });
    
    document.body.appendChild(themeToggle);
    
    themeToggle.addEventListener('click', function() {
        document.body.classList.toggle('dark-mode');
        const isDark = document.body.classList.contains('dark-mode');
        
        this.innerHTML = isDark ? '☀️' : '🌙';
        localStorage.setItem('theme', isDark ? 'dark' : 'light');
        
        // Add transition
        document.body.style.transition = 'background-color 0.3s ease, color 0.3s ease';
    });
    
    // Add dark mode CSS
    const darkModeCSS = `
        body.dark-mode {
            background-color: #1a1a1a;
            color: #f0f0f0;
        }
        
        body.dark-mode .city-card,
        body.dark-mode .highlight-card,
        body.dark-mode .value-card,
        body.dark-mode .info-card {
            background-color: #2a2a2a;
            color: #f0f0f0;
        }
        
        body.dark-mode .footer {
            background: linear-gradient(135deg, #111, #222);
        }
    `;
    
    const style = document.createElement('style');
    style.textContent = darkModeCSS;
    document.head.appendChild(style);
}

// ================= LOCAL STORAGE FUNCTIONS =================
function saveSubscriber(email) {
    const subscribers = JSON.parse(localStorage.getItem('subscribers')) || [];
    subscribers.push({
        email,
        date: new Date().toISOString()
    });
    localStorage.setItem('subscribers', JSON.stringify(subscribers));
}

function saveContactMessage(data) {
    const messages = JSON.parse(localStorage.getItem('contactMessages')) || [];
    messages.push(data);
    localStorage.setItem('contactMessages', JSON.stringify(messages));
}

function saveTripPlan(data) {
    const trips = JSON.parse(localStorage.getItem('tripPlans')) || [];
    trips.push({
        ...data,
        date: new Date().toISOString(),
        id: Date.now()
    });
    localStorage.setItem('tripPlans', JSON.stringify(trips));
}
